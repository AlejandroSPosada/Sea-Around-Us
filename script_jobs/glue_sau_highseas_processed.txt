import sys
from pyspark.context import SparkContext
from awsglue.context import GlueContext
from awsglue.job import Job
from pyspark.sql.functions import col

sc = SparkContext.getOrCreate()
glueContext = GlueContext(sc)
spark = glueContext.spark_session

# Rutas de S3 especificadas
input_path = "s3://seaaroundus-data-eafit/raw/highseas/"
output_path = "s3://seaaroundus-data-eafit/processed/highseas/"

# 1. Lectura del CSV desde la capa Raw
df = spark.read.option("header", "true").option("inferSchema", "true").csv(input_path)

# 2. Normalización de nombres de columnas a snake_case
for column in df.columns:
    clean_col = column.strip().lower().replace(" ", "_").replace("-", "_")
    df = df.withColumnRenamed(column, clean_col)

# 3. Casteo de columnas principales
df = df.withColumn("year", col("year").cast("integer")) \
       .withColumn("tonnes", col("tonnes").cast("float")) \
       .withColumn("landed_value", col("landed_value").cast("float"))

# 4. Manejo de valores nulos
df = df.dropna(subset=["year", "tonnes"])
df = df.fillna({"landed_value": 0.0})

# 5. Escritura en formato Parquet particionado por 'year' en la capa Processed
df.write.mode("overwrite").partitionBy("year").parquet(output_path)
